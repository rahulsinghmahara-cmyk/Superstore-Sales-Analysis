
-- SUPERSTORE SALES ANALYSIS
-- Time-Based Analysis

-- 1. Sales by Year
SELECT
    EXTRACT(YEAR FROM order_date) AS order_year,
    ROUND(SUM(sales), 2) AS total_sales
FROM orders
GROUP BY order_year
ORDER BY order_year;


-- 2. Profit by Year
SELECT
    EXTRACT(YEAR FROM order_date) AS order_year,
    ROUND(SUM(profit), 2) AS total_profit
FROM orders
GROUP BY order_year
ORDER BY order_year;


-- 3. Sales and Profit by Year
SELECT
    EXTRACT(YEAR FROM order_date) AS order_year,
    ROUND(SUM(sales), 2) AS total_sales,
    ROUND(SUM(profit), 2) AS total_profit
FROM orders
GROUP BY order_year
ORDER BY order_year;


-- 4. Monthly Sales
SELECT
    DATE_TRUNC('month', order_date) AS sales_month,
    ROUND(SUM(sales), 2) AS total_sales
FROM orders
GROUP BY sales_month
ORDER BY sales_month;


-- 5. Monthly Sales and Profit
SELECT
    DATE_TRUNC('month', order_date) AS sales_month,
    ROUND(SUM(sales), 2) AS total_sales,
    ROUND(SUM(profit), 2) AS total_profit
FROM orders
GROUP BY sales_month
ORDER BY sales_month;


-- 6. Month-over-Month Sales Change
WITH monthly_sales AS (
    SELECT
        DATE_TRUNC('month', order_date) AS sales_month,
        SUM(sales) AS total_sales
    FROM orders
    GROUP BY sales_month
)
SELECT
    sales_month,
    ROUND(total_sales, 2) AS total_sales,
    ROUND(
        total_sales - LAG(total_sales) OVER (
            ORDER BY sales_month
        ),
        2
    ) AS sales_change
FROM monthly_sales
ORDER BY sales_month;


-- 7. Year-over-Year Sales Change
WITH yearly_sales AS (
    SELECT
        EXTRACT(YEAR FROM order_date) AS order_year,
        SUM(sales) AS total_sales
    FROM orders
    GROUP BY order_year
)
SELECT
    order_year,
    ROUND(total_sales, 2) AS total_sales,
    ROUND(
        total_sales - LAG(total_sales) OVER (
            ORDER BY order_year
        ),
        2
    ) AS sales_change
FROM yearly_sales
ORDER BY order_year;


-- 8. Year-over-Year Sales Percentage Change

    SELECT
        EXTRACT(YEAR FROM order_date) AS order_year,
        SUM(sales) AS total_sales
    FROM orders
    GROUP BY order_year
